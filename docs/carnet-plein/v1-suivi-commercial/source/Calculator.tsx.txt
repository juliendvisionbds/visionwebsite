'use client';

import { useState } from "react";
import { euro } from "@/content/carnet-plein/meta";

const FIELDS = [
  { key: "leads", label: "Demandes reçues par mois", hint: "appels, formulaires, messages", unit: "", max: 5000 },
  { key: "booked", label: "Rendez-vous pris sur ces demandes", hint: "visites, métrés, devis sur place", unit: "", max: 5000 },
  { key: "close", label: "Rendez-vous qui deviennent un chantier", hint: "sur cent", unit: "%", max: 100 },
  { key: "ticket", label: "Valeur moyenne d’un chantier", hint: "hors taxes", unit: "€", max: 1000000 },
] as const;

type Key = (typeof FIELDS)[number]["key"];

/** Ce que vaut un rendez-vous de plus, calculé sur les chiffres du visiteur. `install` : prix de l'installation. */
export default function Calculator({ install }: { install: number }) {
  const [values, setValues] = useState<Record<Key, number>>({ leads: 40, booked: 6, close: 33, ticket: 12000 });
  const [extra, setExtra] = useState(2);

  const set = (key: Key, max: number) => (e: React.ChangeEvent<HTMLInputElement>) =>
    setValues((v) => ({ ...v, [key]: Math.min(max, Math.max(0, Number(e.target.value) || 0)) }));

  const { leads, close, ticket } = values;
  const booked = Math.min(values.booked, leads);
  const lost = leads - booked;
  const perBooking = (close / 100) * ticket;
  const perYear = extra * 12 * perBooking;
  const payback = perBooking > 0 ? Math.ceil(install / perBooking) : null;

  return (
    <div className="calc">
      <div className="calc-in">
        {FIELDS.map((f) => (
          <label key={f.key}>
            <span>
              {f.label}
              <small>{f.hint}</small>
            </span>
            <span className="fill">
              <input type="number" inputMode="numeric" min={0} max={f.max} value={values[f.key] || ""} onChange={set(f.key, f.max)} />
              <i>{f.unit}</i>
            </span>
          </label>
        ))}
        <label className="range">
          <span>
            Rendez-vous en plus par mois
            <small>une hypothèse, à vous de la régler</small>
          </span>
          <span className="fill">
            <input type="range" min={1} max={8} value={extra} onChange={(e) => setExtra(Number(e.target.value))} />
            <i>+{extra}</i>
          </span>
        </label>
      </div>

      <div className="calc-out" aria-live="polite">
        <div className="card-head">
          <b>Votre calcul</b>
          <span className="badge">Estimation</span>
        </div>
        <dl>
          <div>
            <dt>Demandes qui repartent sans rendez-vous</dt>
            <dd>
              {lost} sur {leads}
            </dd>
          </div>
          <div>
            <dt>Ce que vaut un rendez-vous de plus</dt>
            <dd>{euro(perBooking)}</dd>
          </div>
          <div className="total">
            <dt>
              {extra} rendez-vous de plus par mois, sur un an
            </dt>
            <dd>{euro(perYear)}</dd>
          </div>
        </dl>
        <p>
          {payback === null
            ? "Renseignez la valeur d’un chantier et la part de rendez-vous signés."
            : `L’installation (${euro(install)} HT) représente ${payback} rendez-vous supplémentaire${payback > 1 ? "s" : ""}. Chiffre d’affaires, pas marge : à vous d’appliquer la vôtre.`}
        </p>
      </div>
    </div>
  );
}
