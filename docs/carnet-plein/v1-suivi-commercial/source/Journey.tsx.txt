'use client';

import { useRef, useState } from "react";
import { SCENES, fr } from "@/content/carnet-plein/meta";

/** Le parcours en quatre étapes : des onglets, et pour l'étape choisie son explication et un exemple. */
export default function Journey() {
  const [step, setStep] = useState(0);
  const tabs = useRef<Array<HTMLButtonElement | null>>([]);
  const scene = SCENES[step];

  // Flèches, Début et Fin déplacent la sélection, comme pour tout groupe d'onglets.
  function onKeyDown(e: React.KeyboardEvent, i: number) {
    const last = SCENES.length - 1;
    const next = { ArrowRight: i === last ? 0 : i + 1, ArrowLeft: i === 0 ? last : i - 1, Home: 0, End: last }[e.key];
    if (next === undefined) return;
    e.preventDefault();
    setStep(next);
    tabs.current[next]?.focus();
  }

  return (
    <>
      <div className="step-tabs" role="tablist" aria-label="Étapes du parcours">
        {SCENES.map((s, i) => (
          <button
            key={s.tab}
            ref={(el) => {
              tabs.current[i] = el;
            }}
            className={i === step ? "active" : undefined}
            type="button"
            role="tab"
            id={`etape-${i}`}
            aria-selected={i === step}
            aria-controls="etape-panneau"
            tabIndex={i === step ? 0 : -1}
            onClick={() => setStep(i)}
            onKeyDown={(e) => onKeyDown(e, i)}
          >
            <b>{String(i + 1).padStart(2, "0")}</b>
            {s.tab}
          </button>
        ))}
      </div>
      <div className="demo" id="etape-panneau" role="tabpanel" aria-labelledby={`etape-${step}`} tabIndex={0}>
        <div>
          <h3>{fr(scene.title)}</h3>
          <p>{fr(scene.copy)}</p>
        </div>
        <div className="demo-surface">
          <span className="badge brand">{scene.pill}</span>
          <h4>{fr(scene.heading)}</h4>
          <p>
            {scene.detail.map((line) => (
              <span key={line}>{fr(line)}</span>
            ))}
          </p>
        </div>
      </div>
      <p className="demo-foot">Parcours illustratif · configuration adaptée à votre entreprise</p>
    </>
  );
}
